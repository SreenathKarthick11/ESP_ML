About ESP_ML

ESP_ML is a lightweight machine learning client library designed for the ESP32 microcontroller platform. It enables seamless communication between edge devices and cloud-hosted ML models via REST APIs. With support for real-time data acquisition, buffering, and secure API-based requests, ESP_ML allows developers to collect sensor data, send it to a remote server for inference, and receive predictions efficiently. This project bridges the gap between low-power embedded systems and powerful ML models, making intelligent edge computing accessible and scalable for IoT applications.

📘 ESP_ML_Client Library Documentation

This section explains the functions available in the ESP_ML_Client library used in your ESP32 sketch (.ino file).

ESP_ML_Client(int n_features, int max_samples = 100)

Description: Constructor to initialize a new client instance.

  • n_features: Number of input features per sample.
  • max_samples: (Optional) Max number of samples to buffer.

void appendSample(float* features, float label)

Description: Appends a training sample to the buffer.

  • features: Array of input values.
  • label: Expected class label (0 or 1).

String createJsonPayload()

Description: Converts all buffered samples to JSON format for server training.

Returns: JSON string like { "X": [[...]], "y": [...] }

bool sendDataToServer(const char* url, const String& payload)

Description: Sends the training data to the backend server.

Returns: true if upload is successful.

bool getWeights(const char* url, float* weights, float& bias)

Description: Downloads model weights and bias from the server.

Returns: true if data is successfully parsed.

float predict(float* features, float* weights, float bias)

Description: Makes a logistic regression prediction.

Returns: Probability between 0.0 and 1.0.

int predict_class(float* features, float* weights, float bias)

Description: Predicts class based on probability threshold of 0.5.

Returns: 0 or 1.

int getServerPrediction(const char* url, float* features)

Description: Sends a sample to the server and gets predicted class.

Returns: 0, 1 or -1 on failure.

bool setModelType(const char* url, const char* model_type, int max_depth)

Description: Tells the backend what model to use (e.g., "logistic" or "tree").

Returns: true if successful.

void setApiKey(const char* key)

Description: Sets the API key header for authentication.

Parameter: key – API key string provided during registration.

Arduino Example: ESP ML Client Full Flow

#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <esp_ml_client.h>  // Your library

const char* ssid = "Your SSID";
const char* password = ""Your Wifi password";
const char* base_url = "https://esp-ml-backend.onrender.com";
const char* hardcoded_api_key = "Your API KEY "; 

ESP_ML_Client ml_client(3);  // 3 features

void connectWiFi() {
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\\n WiFi Connected!");
}

void trainAndPredict(const char* model_type, int max_depth = 3) {
  Serial.println("\\n==============================");
  Serial.print(" Setting model: ");
  Serial.println(model_type);

  String set_model_url = String(base_url) + "/set_model";
  if (ml_client.setModelType(set_model_url.c_str(), model_type, max_depth)) {
    Serial.println("✅ Model type set successfully");
  } else {
    Serial.println("❌ Failed to set model");
    return;
  }

  float sample1[3] = {1.0, 0.5, -0.3};
  float sample2[3] = {0.7, -0.2, 0.9};
  ml_client.appendSample(sample1, 1.0);
  ml_client.appendSample(sample2, 0.0);

  String upload_url = String(base_url) + "/upload_data";
  String json_payload = ml_client.createJsonPayload();
  if (ml_client.sendDataToServer(upload_url.c_str(), json_payload)) {
    Serial.println("✅ Training data sent");
  } else {
    Serial.println("❌ Training data upload failed");
    return;
  }

  delay(1000);

  float test_sample[3] = {0.8, -0.1, 0.4};
  String predict_url = String(base_url) + "/predict_class";
  int predicted = ml_client.getServerPrediction(predict_url.c_str(), test_sample);

  if (predicted != -1) {
    Serial.print("🎯 Predicted class: ");
    Serial.println(predicted);
  } else {
    Serial.println("❌ Prediction failed");
  }
}

void setup() {
  Serial.begin(115200);
  connectWiFi();
  ml_client.setApiKey(hardcoded_api_key);
  trainAndPredict("logistic");
  delay(2000);
  trainAndPredict("tree", 4);
}

void loop() {}
